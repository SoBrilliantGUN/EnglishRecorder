用法

// 从桶文件统一导入
import { SearchIcon, CloseIcon, SettingsIcon } from '../icons';

// 使用
<SearchIcon />                      // 默认 24px
<CloseIcon size={16} />             // 指定尺寸
<SettingsIcon className={styles.x} /> // 传入样式类

Props（所有图标一致）

Prop类型默认值说明
sizenumber24图标边长（px），同时设置 width / height
classNamestring—外部样式类，可覆盖颜色、尺寸等

颜色继承父级 color（stroke="currentColor"），无需额外传色值。

// 1. 新建 app/src/components/icons/XxxIcon.tsx
interface Props { size?: number; className?: string; }
export default function XxxIcon({ size = 24, className }: Props) {
  return (
    <svg width={size} height={size} viewBox="0 0 24 24"
      fill="none" stroke="currentColor" strokeWidth="2"
      strokeLinecap="round" strokeLinejoin="round"
      className={className}
    >
      <path d="..." />
    </svg>
  );
}

// 2. 在 app/src/components/icons/index.ts 追加导出
export { default as XxxIcon } from './XxxIcon';
